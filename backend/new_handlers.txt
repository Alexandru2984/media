// Update User Profile
pub async fn update_profile(
    State(pool): State<SqlitePool>,
    headers: HeaderMap,
    Json(payload): Json<UpdateUser>,
) -> Result<Json<User>, StatusCode> {
    let user_id = get_current_user(&headers).await?;
    let now = Utc::now();

    // Build dynamic update query
    let mut updates = Vec::new();
    let mut bindings = Vec::new();

    if let Some(username) = &payload.username {
        updates.push("username = ?");
        bindings.push(username.clone());
    }
    if let Some(bio) = &payload.bio {
        updates.push("bio = ?");
        bindings.push(bio.clone());
    }
    if let Some(avatar_url) = &payload.avatar_url {
        updates.push("avatar_url = ?");
        bindings.push(avatar_url.clone());
    }

    if updates.is_empty() {
        return Err(StatusCode::BAD_REQUEST);
    }

    updates.push("updated_at = ?");

    let query_str = format!("UPDATE users SET {} WHERE id = ?", updates.join(", "));
    let mut q = query(&query_str);

    for binding in bindings {
        q = q.bind(binding);
    }
    q = q.bind(now).bind(&user_id);

    q.execute(&pool).await.map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    // Return updated user
    let user = query_as::<_, User>("SELECT * FROM users WHERE id = ?")
        .bind(&user_id)
        .fetch_one(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    Ok(Json(user))
}

// Update Post
pub async fn update_post(
    State(pool): State<SqlitePool>,
    Path(post_id): Path<String>,
    headers: HeaderMap,
    Json(payload): Json<UpdatePost>,
) -> Result<Json<Post>, StatusCode> {
    let user_id = get_current_user(&headers).await?;
    let now = Utc::now();

    // Check ownership
    let post = query_as::<_, Post>("SELECT * FROM posts WHERE id = ?")
        .bind(&post_id)
        .fetch_optional(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?
        .ok_or(StatusCode::NOT_FOUND)?;

    if post.user_id != user_id {
        return Err(StatusCode::FORBIDDEN);
    }

    query("UPDATE posts SET content = ?, image_url = ?, updated_at = ? WHERE id = ?")
        .bind(&payload.content)
        .bind(&payload.image_url)
        .bind(&now)
        .bind(&post_id)
        .execute(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let updated_post = query_as::<_, Post>("SELECT * FROM posts WHERE id = ?")
        .bind(&post_id)
        .fetch_one(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    Ok(Json(updated_post))
}

// Send Message
pub async fn send_message(
    State(pool): State<SqlitePool>,
    headers: HeaderMap,
    Json(payload): Json<CreateMessage>,
) -> Result<Json<Message>, StatusCode> {
    let sender_id = get_current_user(&headers).await?;
    let message_id = Uuid::new_v4().to_string();
    let now = Utc::now();

    let message = query_as::<_, Message>(
        "INSERT INTO messages (id, sender_id, receiver_id, content, created_at) 
         VALUES (?, ?, ?, ?, ?) 
         RETURNING *"
    )
    .bind(&message_id)
    .bind(&sender_id)
    .bind(&payload.receiver_id)
    .bind(&payload.content)
    .bind(&now)
    .fetch_one(&pool)
    .await
    .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    Ok(Json(message))
}

// Get Messages (conversations)
pub async fn get_messages(
    State(pool): State<SqlitePool>,
    Query(params): Query<HashMap<String, String>>,
    headers: HeaderMap,
) -> Result<Json<Vec<MessageWithUser>>, StatusCode> {
    let user_id = get_current_user(&headers).await?;
    let other_user_id = params.get("user_id").ok_or(StatusCode::BAD_REQUEST)?;

    let messages = query_as::<_, Message>(
        "SELECT * FROM messages 
         WHERE (sender_id = ? AND receiver_id = ?) OR (sender_id = ? AND receiver_id = ?)
         ORDER BY created_at ASC"
    )
    .bind(&user_id)
    .bind(other_user_id)
    .bind(other_user_id)
    .bind(&user_id)
    .fetch_all(&pool)
    .await
    .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let mut result = Vec::new();

    for message in messages {
        let sender = query_as::<_, User>("SELECT * FROM users WHERE id = ?")
            .bind(&message.sender_id)
            .fetch_one(&pool)
            .await
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

        result.push(MessageWithUser {
            id: message.id,
            sender_id: message.sender_id,
            receiver_id: message.receiver_id,
            content: message.content,
            read: message.read,
            created_at: message.created_at,
            sender_username: sender.username,
            sender_avatar: sender.avatar_url,
        });
    }

    // Mark messages as read
    query("UPDATE messages SET read = 1 WHERE receiver_id = ? AND sender_id = ?")
        .bind(&user_id)
        .bind(other_user_id)
        .execute(&pool)
        .await
        .ok();

    Ok(Json(result))
}

// Get User Posts
pub async fn get_user_posts(
    State(pool): State<SqlitePool>,
    Path(user_id): Path<String>,
    headers: HeaderMap,
) -> Result<Json<Vec<PostWithUser>>, StatusCode> {
    let current_user_id = get_current_user(&headers).await.ok();

    let posts = query_as::<_, Post>(
        "SELECT * FROM posts WHERE user_id = ? ORDER BY created_at DESC"
    )
    .bind(&user_id)
    .fetch_all(&pool)
    .await
    .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let mut result = Vec::new();

    let user = query_as::<_, User>("SELECT * FROM users WHERE id = ?")
        .bind(&user_id)
        .fetch_one(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    let user_posts_count: i64 = query_scalar("SELECT COUNT(*) FROM posts WHERE user_id = ?")
        .bind(&user_id)
        .fetch_one(&pool)
        .await
        .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;

    for post in posts {
        let is_liked = if let Some(ref uid) = current_user_id {
            let count: i64 = query_scalar(
                "SELECT COUNT(*) FROM likes WHERE post_id = ? AND user_id = ?"
            )
            .bind(&post.id)
            .bind(uid)
            .fetch_one(&pool)
            .await
            .map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
            count > 0
        } else {
            false
        };

        result.push(PostWithUser {
            id: post.id,
            content: post.content,
            image_url: post.image_url,
            likes_count: post.likes_count,
            comments_count: post.comments_count,
            created_at: post.created_at,
            user: UserProfile {
                id: user.id.clone(),
                username: user.username.clone(),
                bio: user.bio.clone(),
                avatar_url: user.avatar_url.clone(),
                posts_count: user_posts_count,
                created_at: user.created_at,
            },
            is_liked,
        });
    }

    Ok(Json(result))
}
